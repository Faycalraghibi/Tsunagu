# Tsunagu MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking. Written to be executed by Claude Sonnet (`claude-sonnet-5-5`).

**Goal:** Build and run a one-day prototype that compares dense RAG, hybrid RAG (BM25 + dense, RRF) and GraphRAG on the LangGraph documentation, then produce `results/results.md` with retrieval metrics, judged correctness with 95% Wilson intervals, faithfulness, citation precision, abstention, latency, tokens and judge calibration.

**Architecture:** A typer CLI (`tsunagu`) drives a linear pipeline: fetch pinned MDX pages, clean, heading-aware chunking with symbols and links, then three indexes (Chroma dense, BM25 + dense fused with RRF, and a JSON property graph built from deterministic links and symbols plus schema-guided LLM triplets with entity resolution). Every retriever returns at most 8 chunks to the same answer prompt. An LLM judge, calibrated on the owner's labels, scores the answers. Every LLM call goes through one cached, rate-limited client, so runs are reproducible and token counts come from the cache.

**Tech Stack:** Python 3.12, typer, pydantic v2, PyYAML, tiktoken, httpx, openai SDK (pointed at NVIDIA NIM), sentence-transformers (`BAAI/bge-small-en-v1.5`, CPU), chromadb, rank-bm25, llama-index-core (`SchemaLLMPathExtractor`), numpy, pytest, ruff, mypy, pre-commit, GitHub Actions.

**Read first:** `docs/graphrag_prototype_spec.md` (the owner's spec; after Task 1 it lives there) and `docs/superpowers/specs/2026-10-22-tsunagu-design.md` (decisions that override or complete the spec).

### 0.5 Budget (NVIDIA NIM, about 35 requests per minute)

| Stage | Calls (estimate) | Wall time at 35 RPM |
|---|---|---|
| Extraction spike (Task 16) | 20 | 1 min |
| Extraction, all chunks (Task 17) | 732 (measured in a dry run of ingest) | about 21 min |
| Answers: 30 questions x 3 systems x 3 repeats | 270 (+ retries) | about 8 min |
| Judge: correctness + faithfulness on answerable, non-abstained answers | up to 486 | about 14 min |

Every call is cached in `cache/llm/`, so re-running any command costs nothing. Run the long stages in the background (`run_in_background`) and do not poll in a sleep loop.

"""Small JSON and JSONL helpers shared by every stage."""

from __future__ import annotations

import json
from pathlib import Path
from typing import Any


def write_json(path: Path, data: Any) -> None:
    """Write `data` as pretty UTF-8 JSON, creating parent folders."""
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(data, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")


def read_json(path: Path) -> Any:
    """Read a UTF-8 JSON file."""
    return json.loads(path.read_text(encoding="utf-8"))


def update_json(path: Path, values: dict[str, Any]) -> dict[str, Any]:
    """Merge `values` into the JSON object stored at `path` (created if missing)."""
    data: dict[str, Any] = read_json(path) if path.exists() else {}
    data.update(values)
    write_json(path, data)
    return data


def write_jsonl(path: Path, rows: list[dict[str, Any]]) -> None:
    """Write one JSON object per line."""
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("w", encoding="utf-8") as f:
        for row in rows:
            f.write(json.dumps(row, ensure_ascii=False) + "\n")


def read_jsonl(path: Path) -> list[dict[str, Any]]:
    """Read a JSONL file, skipping blank lines."""
    with path.open(encoding="utf-8") as f:
        return [json.loads(line) for line in f if line.strip()]
```

```python
"""Download the pinned documentation pages from GitHub."""

from __future__ import annotations

from pathlib import Path

import httpx

RAW_URL = "https://raw.githubusercontent.com/{repo}/{commit}/{path}"


def fetch_pages(
    repo: str,
    commit: str,
    pages: list[str],
    raw_dir: Path,
    *,
    client: httpx.Client | None = None,
    force: bool = False,
) -> list[Path]:
    """Download each page at the pinned commit into `raw_dir`, keeping repo paths.

    Args:
        repo: GitHub "owner/name".
        commit: Full commit hash.
        pages: Repo-relative file paths.
        raw_dir: Destination root.
        client: Optional httpx client (tests inject a mock transport).
        force: Re-download files that already exist.

    Returns:
        The local paths, in the order of `pages`.
    """
    own_client = client is None
    http = client or httpx.Client(timeout=30.0, follow_redirects=True)
    written: list[Path] = []
    try:
        for repo_path in pages:
            dest = raw_dir / repo_path
            if dest.exists() and not force:
                written.append(dest)
                continue
            response = http.get(RAW_URL.format(repo=repo, commit=commit, path=repo_path))
            response.raise_for_status()
            dest.parent.mkdir(parents=True, exist_ok=True)
            dest.write_text(response.text, encoding="utf-8")
            written.append(dest)
    finally:
        if own_client:
            http.close()
    return written
```

```python
"""Turn Mintlify MDX into plain Markdown (Python variant only)."""

from __future__ import annotations

import re
from dataclasses import dataclass

FENCE_RE = re.compile(r"^\s*(```|~~~)")
FRONT_MATTER_RE = re.compile(r"\A---\n(.*?)\n---\n", re.S)
COMMENT_RE = re.compile(r"\{/\*.*?\*/\}", re.S)
API_REF_RE = re.compile(r"@\[`?([^`\]]+?)`?\](?:\[[^\]]*\])?")
JSX_LINE_RE = re.compile(r"^\s*</?(?:[A-Z][A-Za-z0-9]*|a)\b[^>]*>\s*$")
TITLE_ATTR_RE = re.compile(r'title="([^"]+)"')
IMPORT_RE = re.compile(r"^import\s+.+\s+from\s+['\"].+['\"];?$")
H1_RE = re.compile(r"^#\s+(.+?)\s*$", re.M)


@dataclass
class CleanPage:
    title: str
    text: str
    api_refs: list[str]


def _front_matter_title(block: str) -> str:
    for line in block.splitlines():
        if line.startswith("title:"):
            return line.split(":", 1)[1].strip().strip("\"'")
    return ""


def clean_mdx(raw: str) -> CleanPage:
    """Clean one MDX page.

    Drops front matter (keeping the title), comments, JSX tag lines, snippet imports and
    `:::js` blocks; unwraps `:::python` blocks; turns `@[`X`]` API references into code
    spans. Code fences are copied unchanged.

    Args:
        raw: The MDX source.

    Returns:
        The page title, the cleaned Markdown, and the sorted API reference names.
    """
    raw = raw.replace("\r\n", "\n")
    title = ""
    match = FRONT_MATTER_RE.match(raw)
    if match:
        title = _front_matter_title(match.group(1))
        raw = raw[match.end() :]
    raw = COMMENT_RE.sub("", raw)

    out: list[str] = []
    in_fence = False
    lang_block: str | None = None
    for line in raw.split("\n"):
        stripped = line.strip()
        if FENCE_RE.match(line):
            in_fence = not in_fence
            if lang_block != "js":
                out.append(line)
            continue
        if in_fence:
            if lang_block != "js":
                out.append(line)
            continue
        if stripped in (":::python", ":::js"):
            lang_block = stripped[3:]
            continue
        if stripped == ":::":  # closes a language block; a stray closer is dropped too
            lang_block = None
            continue
        if lang_block == "js" or IMPORT_RE.match(stripped):
            continue
        if JSX_LINE_RE.match(line):
            attr = TITLE_ATTR_RE.search(line)
            if attr and not stripped.startswith("</"):
                out.append(f"**{attr.group(1)}**")
            continue
        out.append(line)

    text = "\n".join(out)
    api_refs = sorted({m.strip() for m in API_REF_RE.findall(text)}, key=str.lower)
    text = API_REF_RE.sub(lambda m: f"`{m.group(1).strip()}`", text)
    text = re.sub(r"\n{3,}", "\n\n", text).strip() + "\n"
    if not title:
        h1 = H1_RE.search(text)
        title = h1.group(1) if h1 else ""
    return CleanPage(title=title, text=text, api_refs=api_refs)
```

"""Page ids and internal link normalisation."""

from __future__ import annotations

import re
from dataclasses import dataclass

LINK_RE = re.compile(r"(?<!!)\[[^\]]*\]\(([^)\s]+)(?:\s+\"[^\"]*\")?\)")
LANG_SEGMENTS = {"python", "javascript"}
EXT_RE = re.compile(r"\.(mdx|md)$")


@dataclass
class LinkStats:
    internal: int = 0
    external: int = 0
    out_of_corpus: int = 0


def page_id_from_repo_path(repo_path: str) -> str:
    """Map `src/oss/langgraph/persistence.mdx` to `langgraph/persistence`."""
    page = EXT_RE.sub("", repo_path.removeprefix("src/oss/"))
    return page.removesuffix("/index")


def normalize_link(href: str, source_page: str, known_pages: set[str]) -> tuple[str, str | None]:
    """Classify one link target.

    Returns:
        (kind, target) where kind is "internal", "external", "out_of_corpus" or "anchor",
        and target is a known page id for internal links, else None.
    """
    if href.startswith(("http://", "https://", "mailto:")):
        return "external", None
    if href.startswith("#"):
        return "anchor", None
    path = href.split("#", 1)[0].split("?", 1)[0]
    if path.startswith("/"):
        parts = [p for p in path.split("/") if p]
        if not parts or parts[0] != "oss":
            return "out_of_corpus", None
        parts = parts[1:]
    else:
        resolved: list[str] = source_page.split("/")[:-1]
        for part in path.split("/"):
            if part == "..":
                if resolved:
                    resolved.pop()
            elif part and part != ".":
                resolved.append(part)
        parts = resolved
    parts = [p for p in parts if p not in LANG_SEGMENTS]
    target = EXT_RE.sub("", "/".join(parts)).removesuffix("/index")
    if target in known_pages:
        return "internal", target
    return "out_of_corpus", None


def extract_links(text: str, source_page: str, known_pages: set[str]) -> tuple[list[str], LinkStats]:
    """Find Markdown links in `text` (images excluded).

    Self-links are not returned as targets but are counted as internal.

    Returns:
        Unique internal target page ids in order of first appearance, and link counts.
    """
    targets: list[str] = []
    stats = LinkStats()
    for href in LINK_RE.findall(text):
        kind, target = normalize_link(href, source_page, known_pages)
        if kind == "internal" and target is not None:
            stats.internal += 1
            if target != source_page:
                targets.append(target)
        elif kind == "external":
            stats.external += 1
        elif kind == "out_of_corpus":
            stats.out_of_corpus += 1
    return list(dict.fromkeys(targets)), stats
```

```python
"""Heading-aware chunking that never splits a code block."""

from __future__ import annotations

import json
import re
from dataclasses import asdict, dataclass, field
from pathlib import Path

from tsunagu.config import ChunkingConfig
from tsunagu.links import extract_links
from tsunagu.tokens import count_tokens

FENCE_RE = re.compile(r"^\s*(```|~~~)")
HEADING_RE = re.compile(r"^(#{1,6})\s+(.+?)\s*#*\s*$")
CODE_SPAN_RE = re.compile(r"`([^`\n]+)`")
IDENT_RE = re.compile(r"^[A-Za-z_][A-Za-z0-9_]*(?:\.[A-Za-z_][A-Za-z0-9_]*)*$")
IMPORT_RE = re.compile(r"^\s*from\s+[\w.]+\s+import\s+\(?([\w,\s]+?)\)?\s*$", re.M)
SENTENCE_RE = re.compile(r"(?<=[.!?])\s+|\n")
STOP_SYMBOLS = {
    "true", "false", "none", "null", "str", "int", "float", "bool", "dict", "list",
    "self", "json", "python", "typescript", "javascript", "yes", "the",
}


@dataclass
class Chunk:
    chunk_id: str
    page_path: str
    heading_path: list[str]
    text: str
    n_tokens: int
    symbols: list[str] = field(default_factory=list)
    out_links: list[str] = field(default_factory=list)

    @property
    def anchor(self) -> str:
        """Citation label `page_path#last heading`."""
        heading = self.heading_path[-1] if self.heading_path else ""
        return f"{self.page_path}#{heading}"


@dataclass
class Block:
    text: str
    is_code: bool
    n_tokens: int


def normalize_symbol(raw: str) -> str | None:
    """Turn a code span into a symbol name, or None if it is not an identifier."""
    s = raw.strip().lstrip("@").removesuffix("()")
    if not IDENT_RE.match(s):
        return None
    s = s.split(".")[-1]
    if len(s) < 3 or s.lower() in STOP_SYMBOLS:
        return None
    return s


def _split_prose_code(text: str) -> tuple[str, str]:
    prose: list[str] = []
    code: list[str] = []
    in_fence = False
    for line in text.split("\n"):
        if FENCE_RE.match(line):
            in_fence = not in_fence
            continue
        (code if in_fence else prose).append(line)
    return "\n".join(prose), "\n".join(code)


def extract_symbols(text: str) -> list[str]:
    """Symbols from prose code spans and from `from x import A, B` lines in code."""
    prose, code = _split_prose_code(text)
    found: set[str] = set()
    for match in CODE_SPAN_RE.finditer(prose):
        sym = normalize_symbol(match.group(1))
        if sym:
            found.add(sym)
    for match in IMPORT_RE.finditer(code):
        for name in match.group(1).split(","):
            sym = normalize_symbol(name.strip().split(" as ")[0])
            if sym:
                found.add(sym)
    return sorted(found, key=str.lower)


def split_blocks(body: str) -> list[Block]:
    """Split a section body into paragraphs and whole code blocks."""
    blocks: list[Block] = []
    buf: list[str] = []
    in_fence = False

    def emit(is_code: bool) -> None:
        text = "\n".join(buf).strip("\n")
        if text.strip():
            blocks.append(Block(text, is_code, count_tokens(text)))
        buf.clear()

    for line in body.split("\n"):
        if FENCE_RE.match(line):
            if not in_fence:
                emit(False)
                buf.append(line)
                in_fence = True
            else:
                buf.append(line)
                emit(True)
                in_fence = False
            continue
        if in_fence:
            buf.append(line)
        elif not line.strip():
            emit(False)
        else:
            buf.append(line)
    emit(in_fence)
    return blocks


def _split_oversized(block: Block, max_tokens: int) -> list[Block]:
    if block.is_code or block.n_tokens <= max_tokens:
        return [block]
    pieces: list[Block] = []
    current: list[str] = []
    for sentence in (s for s in SENTENCE_RE.split(block.text) if s.strip()):
        candidate = " ".join([*current, sentence])
        if current and count_tokens(candidate) > max_tokens:
            text = " ".join(current)
            pieces.append(Block(text, False, count_tokens(text)))
            current = [sentence]
        else:
            current.append(sentence)
    if current:
        text = " ".join(current)
        pieces.append(Block(text, False, count_tokens(text)))
    return pieces


def pack_blocks(blocks: list[Block], cfg: ChunkingConfig) -> list[list[Block]]:
    """Greedy packing up to `target_tokens`, carrying trailing prose as overlap."""
    flat = [piece for b in blocks for piece in _split_oversized(b, cfg.max_tokens)]
    groups: list[list[Block]] = []
    current: list[Block] = []
    size = 0
    for block in flat:
        if current and size + block.n_tokens > cfg.target_tokens:
            groups.append(current)
            tail: list[Block] = []
            tail_size = 0
            for prev in reversed(current):
                if prev.is_code or tail_size + prev.n_tokens > cfg.overlap_tokens:
                    break
                tail.insert(0, prev)
                tail_size += prev.n_tokens
            if len(tail) == len(current):
                tail, tail_size = [], 0
            current, size = tail, tail_size
        current.append(block)
        size += block.n_tokens
    if current:
        groups.append(current)
    return groups


def split_sections(text: str, title: str) -> list[tuple[list[str], str]]:
    """Split Markdown on headings (outside code) into (heading_path, body) pairs."""
    sections: list[tuple[list[str], str]] = []
    stack: list[tuple[int, str]] = []
    path = [title]
    body: list[str] = []
    in_fence = False
    for line in text.split("\n"):
        if FENCE_RE.match(line):
            in_fence = not in_fence
        match = None if in_fence else HEADING_RE.match(line)
        if match is None:
            body.append(line)
            continue
        sections.append((path, "\n".join(body)))
        body = []
        level, heading = len(match.group(1)), match.group(2).strip()
        stack = [(lvl, h) for lvl, h in stack if lvl < level]
        stack.append((level, heading))
        path = [title] + [h for _, h in stack if h != title]
    sections.append((path, "\n".join(body)))
    return [(p, b) for p, b in sections if b.strip()]


def chunk_page(
    page_path: str, title: str, text: str, cfg: ChunkingConfig, known_pages: set[str]
) -> list[Chunk]:
    """Chunk one cleaned page.

    Args:
        page_path: Page id such as `langgraph/persistence`.
        title: Page title, used as the root of every heading path.
        text: Cleaned Markdown.
        cfg: Chunk sizes.
        known_pages: Page ids in the corpus, for link normalisation.

    Returns:
        Chunks in reading order, ids `page_path::000`, `page_path::001`, ...
    """
    chunks: list[Chunk] = []
    for heading_path, body in split_sections(text, title):
        for group in pack_blocks(split_blocks(body), cfg):
            header = " > ".join(heading_path)
            chunk_text = header + "\n\n" + "\n\n".join(b.text for b in group)
            links, _ = extract_links(chunk_text, page_path, known_pages)
            chunks.append(
                Chunk(
                    chunk_id=f"{page_path}::{len(chunks):03d}",
                    page_path=page_path,
                    heading_path=list(heading_path),
                    text=chunk_text,
                    n_tokens=count_tokens(chunk_text),
                    symbols=extract_symbols(chunk_text),
                    out_links=links,
                )
            )
    return chunks


def save_chunks(chunks: list[Chunk], path: Path) -> None:
    """Write chunks as JSONL."""
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("w", encoding="utf-8") as f:
        for chunk in chunks:
            f.write(json.dumps(asdict(chunk), ensure_ascii=False) + "\n")


def load_chunks(path: Path) -> list[Chunk]:
    """Read chunks from JSONL."""
    with path.open(encoding="utf-8") as f:
        return [Chunk(**json.loads(line)) for line in f if line.strip()]
```

```python
"""Ingestion: cleaned pages, chunks, explicit link edges, corpus stats."""

from __future__ import annotations

from dataclasses import asdict
from typing import Any

from tsunagu.chunking import Chunk, chunk_page, save_chunks
from tsunagu.config import Settings
from tsunagu.links import extract_links, page_id_from_repo_path
from tsunagu.mdx import clean_mdx
from tsunagu.tokens import count_tokens
from tsunagu.util import write_json, write_jsonl


def _page_index(pages: list[dict[str, Any]], chunks: list[Chunk]) -> str:
    lines = [
        "# Page index",
        "",
        "Use these page ids in `gold_pages`. Headings are listed to help write questions.",
        "",
    ]
    for page in pages:
        lines.append(f"## {page['page_path']}: {page['title']}")
        lines.append(f"{page['n_tokens']} tokens, links to: {', '.join(page['out_links']) or 'none'}")
        seen: list[str] = []
        for chunk in chunks:
            if chunk.page_path == page["page_path"]:
                heading = " > ".join(chunk.heading_path[1:])
                if heading and heading not in seen:
                    seen.append(heading)
        lines.extend(f"- {h}" for h in seen)
        lines.append("")
    return "\n".join(lines)


def run_ingest(settings: Settings) -> dict[str, Any]:
    """Clean, chunk and link every configured page; write outputs to `processed_dir`.

    Returns:
        The corpus statistics also written to `corpus_stats.json`.
    """
    known = {page_id_from_repo_path(p) for p in settings.corpus.pages}
    pages: list[dict[str, Any]] = []
    chunks: list[Chunk] = []
    edges: list[tuple[str, str]] = []
    external = out_of_corpus = 0
    for repo_path in settings.corpus.pages:
        page_id = page_id_from_repo_path(repo_path)
        raw = (settings.paths.raw_dir / repo_path).read_text(encoding="utf-8")
        clean = clean_mdx(raw)
        title = clean.title or page_id
        targets, stats = extract_links(clean.text, page_id, known)
        external += stats.external
        out_of_corpus += stats.out_of_corpus
        edges.extend((page_id, t) for t in targets)
        page_chunks = chunk_page(page_id, title, clean.text, settings.chunking, known)
        chunks.extend(page_chunks)
        pages.append(
            {
                "page_path": page_id,
                "repo_path": repo_path,
                "title": title,
                "n_tokens": count_tokens(clean.text),
                "n_chunks": len(page_chunks),
                "out_links": targets,
                "api_refs": clean.api_refs,
                "link_stats": asdict(stats),
            }
        )

    out = settings.paths.processed_dir
    write_jsonl(out / "pages.jsonl", pages)
    save_chunks(chunks, out / "chunks.jsonl")
    out.mkdir(parents=True, exist_ok=True)
    (out / "link_edges.csv").write_text(
        "source,target\n" + "".join(f"{s},{t}\n" for s, t in edges), encoding="utf-8"
    )
    (out / "page_index.md").write_text(_page_index(pages, chunks), encoding="utf-8")
    sizes = [c.n_tokens for c in chunks]
    stats_out: dict[str, Any] = {
        "repo": settings.corpus.repo,
        "commit": settings.corpus.commit,
        "n_pages": len(pages),
        "n_tokens": sum(p["n_tokens"] for p in pages),
        "n_chunks": len(chunks),
        "mean_chunk_tokens": round(sum(sizes) / len(sizes), 1) if sizes else 0,
        "max_chunk_tokens": max(sizes, default=0),
        "n_link_edges": len(edges),
        "links_external": external,
        "links_out_of_corpus": out_of_corpus,
        "n_chunks_with_symbols": sum(1 for c in chunks if c.symbols),
    }
    write_json(out / "corpus_stats.json", stats_out)
    return stats_out

"""The owner's question set: model, loader, validator."""

from __future__ import annotations

import json
from collections import Counter
from pathlib import Path
from typing import Literal

from pydantic import BaseModel, model_validator

QuestionType = Literal["single_fact", "multi_hop", "relational", "exact_reference", "unanswerable"]
EXPECTED_COUNTS: dict[str, int] = {
    "single_fact": 7,
    "multi_hop": 8,
    "relational": 8,
    "exact_reference": 4,
    "unanswerable": 3,
}


class Question(BaseModel):
    id: str
    type: QuestionType
    question: str
    gold_answer: str
    gold_pages: list[str]
    notes: str = ""

    @model_validator(mode="after")
    def _pages_match_type(self) -> Question:
        if self.type == "unanswerable" and self.gold_pages:
            raise ValueError("unanswerable questions must have gold_pages = []")
        if self.type != "unanswerable" and not 1 <= len(self.gold_pages) <= 4:
            raise ValueError("answerable questions need 1 to 4 gold_pages")
        return self


def load_questions(path: Path) -> list[Question]:
    """Load questions from JSONL (one object per line)."""
    with path.open(encoding="utf-8") as f:
        return [Question.model_validate(json.loads(line)) for line in f if line.strip()]


def validate_questions(questions: list[Question], known_pages: set[str]) -> list[str]:
    """Return human-readable problems; an empty list means the set is ready."""
    problems: list[str] = []
    for qid, n in Counter(q.id for q in questions).items():
        if n > 1:
            problems.append(f"duplicate id {qid}")
    for q in questions:
        problems.extend(f"{q.id}: unknown page {p}" for p in q.gold_pages if p not in known_pages)
    counts: Counter[str] = Counter(q.type for q in questions)
    for qtype, expected in EXPECTED_COUNTS.items():
        if counts.get(qtype, 0) != expected:
            problems.append(f"{qtype}: {counts.get(qtype, 0)} (expected {expected})")
    return problems
```

```python
"""OpenAI-compatible LLM client with a disk cache, a rate limiter and retries."""

from __future__ import annotations

import hashlib
import json
import os
import re
import time
from collections.abc import Callable
from dataclasses import dataclass
from pathlib import Path
from typing import Any

from tsunagu.config import Settings

Transport = Callable[[dict[str, Any]], dict[str, Any]]
THINK_RE = re.compile(r"<think>.*?</think>", re.S)


class RetryableLLMError(RuntimeError):
    """Rate limit, timeout, connection or 5xx error: worth retrying."""


@dataclass
class LLMResponse:
    text: str
    prompt_tokens: int
    completion_tokens: int
    latency_s: float
    cached: bool


class DiskCache:
    """One JSON file per call under `root/<2 hex>/<sha256>.json`."""

    def __init__(self, root: Path) -> None:
        self.root = root

    @staticmethod
    def key(payload: dict[str, Any]) -> str:
        blob = json.dumps(payload, sort_keys=True, ensure_ascii=False)
        return hashlib.sha256(blob.encode("utf-8")).hexdigest()

    def _path(self, key: str) -> Path:
        return self.root / key[:2] / f"{key}.json"

    def get(self, key: str) -> dict[str, Any] | None:
        path = self._path(key)
        if not path.exists():
            return None
        record: dict[str, Any] = json.loads(path.read_text(encoding="utf-8"))
        return record

    def put(self, key: str, record: dict[str, Any]) -> None:
        path = self._path(key)
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_text(json.dumps(record, ensure_ascii=False, indent=1), encoding="utf-8")


class RateLimiter:
    """Space calls at least 60/rpm seconds apart."""

    def __init__(
        self,
        rpm: int,
        clock: Callable[[], float] = time.monotonic,
        sleep: Callable[[float], None] = time.sleep,
    ) -> None:
        self.interval = 60.0 / rpm if rpm > 0 else 0.0
        self.clock = clock
        self.sleep = sleep
        self._next = 0.0

    def wait(self) -> None:
        now = self.clock()
        if now < self._next:
            self.sleep(self._next - now)
            now = self._next
        self._next = now + self.interval


def strip_thinking(text: str) -> str:
    """Remove `<think>...</think>` blocks some reasoning models emit."""
    return THINK_RE.sub("", text).strip()


def extract_json(text: str) -> dict[str, Any]:
    """Parse the first JSON object in `text` (tolerates code fences and chatter).

    Raises:
        ValueError: if no JSON object can be parsed.
    """
    start, end = text.find("{"), text.rfind("}")
    if start == -1 or end <= start:
        raise ValueError("no JSON object found")
    obj = json.loads(text[start : end + 1])
    if not isinstance(obj, dict):
        raise ValueError("JSON value is not an object")
    return obj


class LLMClient:
    """The single entry point for every LLM call in the project."""

    def __init__(
        self,
        model: str,
        transport: Transport,
        cache: DiskCache,
        *,
        limiter: RateLimiter | None = None,
        temperature: float = 0.0,
        max_tokens: int = 2048,
        max_retries: int = 5,
        extra_body: dict[str, Any] | None = None,
        sleep: Callable[[float], None] = time.sleep,
    ) -> None:
        self.model = model
        self.transport = transport
        self.cache = cache
        self.limiter = limiter
        self.temperature = temperature
        self.max_tokens = max_tokens
        self.max_retries = max_retries
        self.extra_body = extra_body or {}
        self.sleep = sleep

    def complete(self, messages: list[dict[str, Any]], *, stage: str, tag: str = "") -> LLMResponse:
        """Return the model reply, from cache when this exact call was made before.

        Args:
            messages: Chat messages.
            stage: Accounting bucket: "extraction", "answer" or "judge".
            tag: Distinguishes otherwise identical calls (for example repeat index).
        """
        payload: dict[str, Any] = {
            "model": self.model,
            "messages": messages,
            "temperature": self.temperature,
            "max_tokens": self.max_tokens,
            "extra_body": self.extra_body,
            "tag": tag,
        }
        key = DiskCache.key(payload)
        hit = self.cache.get(key)
        if hit is not None:
            return LLMResponse(hit["text"], hit["prompt_tokens"], hit["completion_tokens"], hit["latency_s"], True)

        request = {k: v for k, v in payload.items() if k != "tag"}
        for attempt in range(self.max_retries + 1):
            if self.limiter:
                self.limiter.wait()
            started = time.perf_counter()
            try:
                out = self.transport(request)
            except RetryableLLMError:
                if attempt == self.max_retries:
                    raise
                self.sleep(min(60.0, 5.0 * 2**attempt))
                continue
            latency = time.perf_counter() - started
            record: dict[str, Any] = {
                "text": strip_thinking(out["text"]),
                "prompt_tokens": int(out.get("prompt_tokens", 0)),
                "completion_tokens": int(out.get("completion_tokens", 0)),
                "latency_s": latency,
                "stage": stage,
                "model": self.model,
                "tag": tag,
                "created": time.time(),
            }
            self.cache.put(key, record)
            return LLMResponse(record["text"], record["prompt_tokens"], record["completion_tokens"], latency, False)
        raise RuntimeError("unreachable")


def openai_transport(base_url: str, api_key: str, timeout_s: float) -> Transport:
    """Transport that calls an OpenAI-compatible /chat/completions endpoint."""
    from openai import APIConnectionError, APITimeoutError, InternalServerError, OpenAI, RateLimitError

    client = OpenAI(base_url=base_url, api_key=api_key, timeout=timeout_s, max_retries=0)

    def call(request: dict[str, Any]) -> dict[str, Any]:
        try:
            resp = client.chat.completions.create(
                model=request["model"],
                messages=request["messages"],
                temperature=request["temperature"],
                max_tokens=request["max_tokens"],
                extra_body=request["extra_body"] or None,
            )
        except (RateLimitError, APIConnectionError, APITimeoutError, InternalServerError) as exc:
            raise RetryableLLMError(str(exc)) from exc
        usage = resp.usage
        return {
            "text": resp.choices[0].message.content or "",
            "prompt_tokens": usage.prompt_tokens if usage else 0,
            "completion_tokens": usage.completion_tokens if usage else 0,
        }

    return call


def build_llm(settings: Settings) -> LLMClient:
    """Build the real client from settings and the API key in `.env`."""
    from dotenv import load_dotenv

    load_dotenv()
    cfg = settings.llm
    api_key = os.environ.get(cfg.api_key_env, "")
    if not api_key or api_key == "__PLACEHOLDER__":
        raise RuntimeError(f"{cfg.api_key_env} is not set. Add it to .env (see .env.example).")
    extra = {"chat_template_kwargs": {"enable_thinking": False}} if cfg.disable_thinking else {}
    return LLMClient(
        model=cfg.model,
        transport=openai_transport(cfg.base_url, api_key, cfg.timeout_s),
        cache=DiskCache(settings.paths.cache_dir),
        limiter=RateLimiter(cfg.rpm),
        temperature=cfg.temperature,
        max_tokens=cfg.max_tokens,
        max_retries=cfg.max_retries,
        extra_body=extra,
    )


def usage_by_stage(cache_root: Path) -> dict[str, dict[str, float]]:
    """Sum calls, tokens and original latency per stage over every cached call."""
    totals: dict[str, dict[str, float]] = {}
    for path in cache_root.glob("*/*.json"):
        record = json.loads(path.read_text(encoding="utf-8"))
        bucket = totals.setdefault(
            record.get("stage", "unknown"),
            {"calls": 0, "prompt_tokens": 0, "completion_tokens": 0, "latency_s": 0.0},
        )
        bucket["calls"] += 1
        bucket["prompt_tokens"] += record.get("prompt_tokens", 0)
        bucket["completion_tokens"] += record.get("completion_tokens", 0)
        bucket["latency_s"] += record.get("latency_s", 0.0)
    return totals

"""Embedding backends: bge-small via sentence-transformers, and a hash embedder for tests."""

from __future__ import annotations

import hashlib
import re
from typing import Protocol

import numpy as np

BGE_QUERY_PREFIX = "Represent this sentence for searching relevant passages: "


class Embedder(Protocol):
    def embed_documents(self, texts: list[str]) -> np.ndarray: ...

    def embed_query(self, text: str) -> np.ndarray: ...


class SentenceTransformerEmbedder:
    """`BAAI/bge-small-en-v1.5` on CPU, L2-normalised, with the bge query prefix."""

    def __init__(self, model_name: str, batch_size: int = 32) -> None:
        from sentence_transformers import SentenceTransformer

        self.model = SentenceTransformer(model_name, device="cpu")
        self.batch_size = batch_size

    def embed_documents(self, texts: list[str]) -> np.ndarray:
        vectors = self.model.encode(
            texts,
            batch_size=self.batch_size,
            normalize_embeddings=True,
            convert_to_numpy=True,
            show_progress_bar=len(texts) > 200,
        )
        return np.asarray(vectors, dtype=np.float32)

    def embed_query(self, text: str) -> np.ndarray:
        return self.embed_documents([BGE_QUERY_PREFIX + text])[0]


class HashEmbedder:
    """Deterministic bag-of-words hashing embedder (tests only, no downloads)."""

    def __init__(self, dim: int = 256) -> None:
        self.dim = dim

    def _vector(self, text: str) -> np.ndarray:
        vec = np.zeros(self.dim, dtype=np.float32)
        for token in re.findall(r"[a-z0-9]+", text.lower()):
            vec[int(hashlib.md5(token.encode()).hexdigest(), 16) % self.dim] += 1.0
        norm = float(np.linalg.norm(vec))
        return vec / norm if norm else vec

    def embed_documents(self, texts: list[str]) -> np.ndarray:
        if not texts:
            return np.zeros((0, self.dim), dtype=np.float32)
        return np.stack([self._vector(t) for t in texts])

    def embed_query(self, text: str) -> np.ndarray:
        return self._vector(text)
```

        collection.add(
            ids=[c.chunk_id for c in part],
            embeddings=embeddings[start : start + batch].tolist(),
            documents=[c.text for c in part],
            metadatas=[{"page_path": c.page_path} for c in part],

```python
"""Retrievers A (dense) and B (hybrid BM25 + dense with RRF)."""

from __future__ import annotations

import re
import time
from dataclasses import dataclass, field
from typing import Any, Protocol

from rank_bm25 import BM25Okapi

from tsunagu.embeddings import Embedder
from tsunagu.store import ChunkStore

TOKEN_RE = re.compile(r"[A-Za-z0-9_]+")
CAMEL_RE = re.compile(r"[A-Z]?[a-z]+|[A-Z]+(?![a-z])|\d+")


@dataclass
class RetrievedChunk:
    chunk_id: str
    page_path: str
    heading_path: list[str]
    text: str
    score: float

    @property
    def label(self) -> str:
        """Citation label shown to the answering model."""
        heading = self.heading_path[-1] if self.heading_path else ""
        return f"{self.page_path}#{heading}"


@dataclass
class RetrievalResult:
    system: str
    chunks: list[RetrievedChunk]
    latency_s: float
    trace: dict[str, Any] = field(default_factory=dict)


class Retriever(Protocol):
    name: str

    def retrieve(self, query: str, k: int) -> RetrievalResult: ...


def to_retrieved(store: ChunkStore, chunk_id: str, score: float) -> RetrievedChunk:
    chunk = store.get(chunk_id)
    return RetrievedChunk(chunk.chunk_id, chunk.page_path, chunk.heading_path, chunk.text, score)


def bm25_tokenize(text: str) -> list[str]:
    """Lowercase word tokens plus the parts of camelCase and snake_case identifiers."""
    tokens: list[str] = []
    for tok in TOKEN_RE.findall(text):
        low = tok.lower()
        tokens.append(low)
        parts = [p.lower() for piece in tok.split("_") for p in CAMEL_RE.findall(piece)]
        if len(parts) > 1:
            tokens.extend(parts)
    return tokens


def rrf_fuse(rankings: list[list[str]], k: int = 60) -> list[tuple[str, float]]:
    """Reciprocal rank fusion: score(d) = sum over rankings of 1 / (k + rank), rank from 1."""
    scores: dict[str, float] = {}
    for ranking in rankings:
        for rank, doc_id in enumerate(ranking, start=1):
            scores[doc_id] = scores.get(doc_id, 0.0) + 1.0 / (k + rank)
    order = {doc_id: i for i, doc_id in enumerate(scores)}
    return sorted(scores.items(), key=lambda item: (-item[1], order[item[0]]))


class DenseRetriever:
    name = "dense"

    def __init__(self, collection: Any, store: ChunkStore, embedder: Embedder) -> None:
        self.collection = collection
        self.store = store
        self.embedder = embedder

    def ranked_ids(self, query: str, k: int) -> list[tuple[str, float]]:
        vector = self.embedder.embed_query(query)
        res = self.collection.query(query_embeddings=[vector.tolist()], n_results=k)
        return [(cid, 1.0 - float(d)) for cid, d in zip(res["ids"][0], res["distances"][0], strict=True)]

    def retrieve(self, query: str, k: int) -> RetrievalResult:
        started = time.perf_counter()
        ranked = self.ranked_ids(query, k)
        chunks = [to_retrieved(self.store, cid, s) for cid, s in ranked]
        return RetrievalResult(self.name, chunks, time.perf_counter() - started)


class BM25Retriever:
    name = "bm25"

    def __init__(self, store: ChunkStore) -> None:
        self.store = store
        self.bm25 = BM25Okapi([bm25_tokenize(c.text) for c in store.chunks])

    def ranked_ids(self, query: str, k: int) -> list[tuple[str, float]]:
        scores = self.bm25.get_scores(bm25_tokenize(query))
        top = sorted(range(len(scores)), key=lambda i: -scores[i])[:k]
        return [(self.store.ids[i], float(scores[i])) for i in top]

    def retrieve(self, query: str, k: int) -> RetrievalResult:
        started = time.perf_counter()
        chunks = [to_retrieved(self.store, cid, s) for cid, s in self.ranked_ids(query, k)]
        return RetrievalResult(self.name, chunks, time.perf_counter() - started)


class HybridRetriever:
    name = "hybrid"

    def __init__(self, dense: DenseRetriever, bm25: BM25Retriever, candidate_k: int = 20, rrf_k: int = 60) -> None:
        self.dense = dense
        self.bm25 = bm25
        self.candidate_k = candidate_k
        self.rrf_k = rrf_k

    def retrieve(self, query: str, k: int) -> RetrievalResult:
        started = time.perf_counter()
        dense_ids = [cid for cid, _ in self.dense.ranked_ids(query, self.candidate_k)]
        bm25_ids = [cid for cid, _ in self.bm25.ranked_ids(query, self.candidate_k)]
        fused = rrf_fuse([dense_ids, bm25_ids], k=self.rrf_k)[:k]
        chunks = [to_retrieved(self.dense.store, cid, s) for cid, s in fused]
        return RetrievalResult(self.name, chunks, time.perf_counter() - started)
```

"""Build the dense index: chunk embeddings on disk and in Chroma."""

from __future__ import annotations

import json
import time
from typing import Any

import numpy as np

from tsunagu.config import Settings
from tsunagu.embeddings import Embedder
from tsunagu.store import ChunkStore
from tsunagu.util import update_json
from tsunagu.vector_index import add_chunks, create_collection, persistent_client


def build_dense_index(settings: Settings, store: ChunkStore, embedder: Embedder) -> dict[str, Any]:
    """Embed every chunk, save `chunk_embeddings.npy` + `chunk_ids.json`, fill Chroma.

    Returns:
        Timing and size facts, also merged into `results/indexing.json`.
    """
    started = time.perf_counter()
    vectors = embedder.embed_documents([c.text for c in store.chunks])
    index_dir = settings.paths.index_dir
    index_dir.mkdir(parents=True, exist_ok=True)
    np.save(index_dir / "chunk_embeddings.npy", vectors)
    (index_dir / "chunk_ids.json").write_text(json.dumps(store.ids), encoding="utf-8")
    collection = create_collection(persistent_client(index_dir))
    add_chunks(collection, store.chunks, vectors)
    facts = {
        "dense_index_seconds": round(time.perf_counter() - started, 1),
        "embed_model": settings.retrieval.embed_model,
        "n_chunks_indexed": len(store.ids),
        "embedding_dim": int(vectors.shape[1]),
    }
    update_json(settings.paths.results_dir / "indexing.json", facts)
    return facts
```

```python
"""One answer prompt shared by every system: context only, citations, or abstain."""

from __future__ import annotations

from dataclasses import dataclass, field
from typing import Any

from pydantic import BaseModel, ValidationError

from tsunagu.llm import LLMClient, extract_json
from tsunagu.retrievers import RetrievedChunk

INSUFFICIENT = "INSUFFICIENT_CONTEXT"

SYSTEM_PROMPT = f"""You answer questions about the LangGraph documentation using ONLY the context passages provided.
Rules:
1. Use only facts stated in the context. Do not use outside knowledge.
2. Each passage starts with its label in square brackets, like [langgraph/persistence#Checkpointers]. Put the label of every passage that supports your answer in "citations", copied exactly.
3. If the context does not contain the answer, set "answer" to "{INSUFFICIENT}", "citations" to [] and "abstained" to true.
4. Reply with one JSON object and nothing else:
{{"answer": "<2 to 6 sentences>", "citations": ["<label>", "..."], "abstained": false}}"""

RETRY_MESSAGE = "Your reply was not a valid JSON object with keys answer, citations and abstained. Reply again with only that JSON object."


class AnswerPayload(BaseModel):
    answer: str
    citations: list[str] = []
    abstained: bool = False


@dataclass
class Answer:
    answer: str
    citations: list[str]
    abstained: bool
    parse_ok: bool
    prompt_tokens: int = 0
    completion_tokens: int = 0
    latency_s: float = 0.0
    raw: list[str] = field(default_factory=list)


def format_context(chunks: list[RetrievedChunk]) -> str:
    return "\n\n".join(f"[{c.label}]\n{c.text}" for c in chunks)


def build_messages(question: str, chunks: list[RetrievedChunk]) -> list[dict[str, Any]]:
    user = f"Context passages:\n\n{format_context(chunks)}\n\nQuestion: {question}"
    return [{"role": "system", "content": SYSTEM_PROMPT}, {"role": "user", "content": user}]


def cited_pages(citations: list[str]) -> list[str]:
    """Unique page ids from `page#heading` labels, in order."""
    return list(dict.fromkeys(c.split("#", 1)[0].strip().strip("[]") for c in citations if c.strip()))


def answer_question(llm: LLMClient, question: str, chunks: list[RetrievedChunk], *, tag: str) -> Answer:
    """Answer from the given chunks; retry once if the reply is not valid JSON."""
    messages = build_messages(question, chunks)
    total_in = total_out = 0
    latency = 0.0
    raws: list[str] = []
    for attempt in range(2):
        resp = llm.complete(messages, stage="answer", tag=tag if attempt == 0 else f"{tag}:retry")
        total_in += resp.prompt_tokens
        total_out += resp.completion_tokens
        latency += resp.latency_s
        raws.append(resp.text)
        try:
            payload = AnswerPayload.model_validate(extract_json(resp.text))
        except (ValueError, ValidationError):
            messages = [*messages, {"role": "assistant", "content": resp.text}, {"role": "user", "content": RETRY_MESSAGE}]
            continue
        abstained = payload.abstained or payload.answer.strip() == INSUFFICIENT
        citations = [] if abstained else payload.citations
        return Answer(payload.answer, citations, abstained, True, total_in, total_out, latency, raws)
    return Answer(raws[-1], [], False, False, total_in, total_out, latency, raws)
```

"""Retrieval metrics, Wilson intervals, Cohen's kappa, percentiles."""

from __future__ import annotations

import math
from collections import Counter

import numpy as np


def page_ranking(page_paths: list[str]) -> list[str]:
    """Pages in order of first appearance among retrieved chunks."""
    return list(dict.fromkeys(page_paths))


def hit_at_k(pages: list[str], gold: list[str]) -> float:
    return 1.0 if set(pages) & set(gold) else 0.0


def recall_at_k(pages: list[str], gold: list[str]) -> float:
    return len(set(pages) & set(gold)) / len(set(gold)) if gold else 0.0


def mrr(pages: list[str], gold: list[str]) -> float:
    gold_set = set(gold)
    for rank, page in enumerate(pages, start=1):
        if page in gold_set:
            return 1.0 / rank
    return 0.0


def wilson_interval(successes: int, n: int, z: float = 1.96) -> tuple[float, float]:
    """95% Wilson score interval for a binomial proportion."""
    if n == 0:
        return 0.0, 0.0
    p = successes / n
    denom = 1 + z**2 / n
    center = (p + z**2 / (2 * n)) / denom
    half = z * math.sqrt(p * (1 - p) / n + z**2 / (4 * n**2)) / denom
    return max(0.0, center - half), min(1.0, center + half)


def cohen_kappa(a: list[int], b: list[int]) -> float:
    """Unweighted Cohen's kappa between two raters."""
    if len(a) != len(b) or not a:
        raise ValueError("ratings must be non-empty and the same length")
    n = len(a)
    observed = sum(x == y for x, y in zip(a, b, strict=True)) / n
    ca, cb = Counter(a), Counter(b)
    expected = sum(ca[k] * cb[k] for k in set(ca) | set(cb)) / n**2
    if expected == 1.0:
        return 1.0 if observed == 1.0 else 0.0
    return (observed - expected) / (1 - expected)


def percentile(values: list[float], q: float) -> float:
    """Linear-interpolated percentile; NaN for an empty list."""
    return float(np.percentile(values, q)) if values else float("nan")
```

"""A small JSON property graph: deterministic skeleton plus LLM triplets."""

from __future__ import annotations

import csv
from collections import Counter
from dataclasses import asdict, dataclass, field
from pathlib import Path
from typing import Any

from tsunagu.chunking import Chunk
from tsunagu.util import read_json, write_json

ENTITY_TYPES = ("Concept", "Class", "Function", "Parameter", "Feature")
LLM_RELATIONS = ("USES", "REQUIRES", "CONFIGURES", "PART_OF", "ALTERNATIVE_TO", "DEPRECATED_BY", "RETURNS")
STRUCTURAL_RELATIONS = ("LINKS_TO", "MENTIONS")
ALL_RELATIONS = frozenset(LLM_RELATIONS + STRUCTURAL_RELATIONS)


def node_id(type_: str, name: str) -> str:
    return f"{type_}:{name}"


@dataclass
class ExtractedTriplet:
    subject: str
    subject_type: str
    relation: str
    object: str
    object_type: str
    chunk_id: str


@dataclass
class Node:
    id: str
    name: str
    type: str
    title: str = ""
    chunk_ids: set[str] = field(default_factory=set)
    aliases: set[str] = field(default_factory=set)

    @property
    def embed_text(self) -> str:
        """Text embedded for seed search: the page title for pages, else the name."""
        return self.title or self.name

    def to_dict(self) -> dict[str, Any]:
        return {
            "id": self.id,
            "name": self.name,
            "type": self.type,
            "title": self.title,
            "chunk_ids": sorted(self.chunk_ids),
            "aliases": sorted(self.aliases),
        }

    @classmethod
    def from_dict(cls, d: dict[str, Any]) -> Node:
        return cls(d["id"], d["name"], d["type"], d.get("title", ""), set(d["chunk_ids"]), set(d["aliases"]))


@dataclass(frozen=True)
class Edge:
    src: str
    rel: str
    dst: str
    source: str  # "links", "symbols" or "llm"
    chunk_id: str = ""


class Graph:
    def __init__(self) -> None:
        self.nodes: dict[str, Node] = {}
        self._edges: dict[tuple[str, str, str], Edge] = {}
        self._adj: dict[str, list[tuple[Edge, str]]] | None = None

    def add_node(self, name: str, type_: str, *, chunk_id: str | None = None, title: str = "") -> str:
        nid = node_id(type_, name)
        node = self.nodes.get(nid)
        if node is None:
            node = Node(nid, name, type_, title)
            self.nodes[nid] = node
            self._adj = None
        if chunk_id:
            node.chunk_ids.add(chunk_id)
        return nid

    def add_edge(self, src: str, rel: str, dst: str, source: str, chunk_id: str = "") -> None:
        if src == dst:
            return
        key = (src, rel, dst)
        if key not in self._edges:
            self._edges[key] = Edge(src, rel, dst, source, chunk_id)
            self._adj = None

    @property
    def edges(self) -> list[Edge]:
        return list(self._edges.values())

    def _adjacency(self) -> dict[str, list[tuple[Edge, str]]]:
        if self._adj is None:
            adj: dict[str, list[tuple[Edge, str]]] = {}
            for e in self._edges.values():
                adj.setdefault(e.src, []).append((e, e.dst))
                adj.setdefault(e.dst, []).append((e, e.src))
            self._adj = adj
        return self._adj

    def neighbors(self, nid: str, relations: set[str] | frozenset[str]) -> list[tuple[Edge, str]]:
        """Edges touching `nid` in either direction whose relation is in `relations`."""
        return [(e, other) for e, other in self._adjacency().get(nid, []) if e.rel in relations]

    def degree(self, nid: str) -> int:
        return len(self._adjacency().get(nid, []))

    def stats(self) -> dict[str, Any]:
        return {
            "nodes": len(self.nodes),
            "entities": sum(1 for n in self.nodes.values() if n.type != "Page"),
            "llm_entities": sum(1 for n in self.nodes.values() if n.type in ENTITY_TYPES),
            "nodes_by_type": dict(Counter(n.type for n in self.nodes.values())),
            "edges": len(self._edges),
            "llm_relations": sum(1 for e in self._edges.values() if e.rel in LLM_RELATIONS),
            "explicit_links": sum(1 for e in self._edges.values() if e.rel == "LINKS_TO"),
            "edges_by_relation": dict(Counter(e.rel for e in self._edges.values())),
        }

    def save(self, path: Path) -> None:
        write_json(path, {"nodes": [n.to_dict() for n in self.nodes.values()], "edges": [asdict(e) for e in self.edges]})

    @classmethod
    def load(cls, path: Path) -> Graph:
        data = read_json(path)
        g = cls()
        for d in data["nodes"]:
            node = Node.from_dict(d)
            g.nodes[node.id] = node
        for e in data["edges"]:
            g.add_edge(e["src"], e["rel"], e["dst"], e["source"], e.get("chunk_id", ""))
        return g


def load_link_edges(path: Path) -> list[tuple[str, str]]:
    with path.open(encoding="utf-8") as f:
        return [(row["source"], row["target"]) for row in csv.DictReader(f)]


def build_deterministic_graph(
    pages: list[dict[str, Any]], chunks: list[Chunk], link_edges: list[tuple[str, str]]
) -> Graph:
    """Pages, explicit page links and code-symbol mentions: no LLM involved."""
    g = Graph()
    for page in pages:
        g.add_node(page["page_path"], "Page", title=page["title"])
    for chunk in chunks:
        pid = node_id("Page", chunk.page_path)
        g.nodes[pid].chunk_ids.add(chunk.chunk_id)
        for symbol in chunk.symbols:
            sid = g.add_node(symbol, "Symbol", chunk_id=chunk.chunk_id)
            g.add_edge(pid, "MENTIONS", sid, "symbols", chunk.chunk_id)
    for source, target in link_edges:
        g.add_edge(node_id("Page", source), "LINKS_TO", node_id("Page", target), "links")
    return g


def add_triplets(g: Graph, triplets: list[ExtractedTriplet], page_of: dict[str, str]) -> None:
    """Add LLM triplets, plus `Page -MENTIONS-> entity` edges for their source chunk."""
    for t in triplets:
        s = g.add_node(t.subject, t.subject_type, chunk_id=t.chunk_id)
        o = g.add_node(t.object, t.object_type, chunk_id=t.chunk_id)
        g.add_edge(s, t.relation, o, "llm", t.chunk_id)
        page = node_id("Page", page_of[t.chunk_id])
        g.add_edge(page, "MENTIONS", s, "llm", t.chunk_id)
        g.add_edge(page, "MENTIONS", o, "llm", t.chunk_id)
```

"""Schema-guided triplet extraction: LlamaIndex SchemaLLMPathExtractor or our own JSON prompt."""

from __future__ import annotations

from typing import Any, Literal, Protocol

from tsunagu.chunking import Chunk
from tsunagu.graph import ENTITY_TYPES, LLM_RELATIONS, ExtractedTriplet
from tsunagu.llm import LLMClient, extract_json

TYPE_BY_UPPER = {t.upper(): t for t in ENTITY_TYPES}

EXTRACT_PROMPT = """Extract a knowledge graph from this passage of the LangGraph documentation.
Entity types: Concept, Class, Function, Parameter, Feature.
Relation types: USES, REQUIRES, CONFIGURES, PART_OF, ALTERNATIVE_TO, DEPRECATED_BY, RETURNS.
Rules:
- Use names exactly as written in the passage; keep code names such as InMemorySaver unchanged.
- Only extract relations the passage states or directly shows.
- At most {max_triplets} triplets.
Reply with one JSON object only:
{{"triplets": [{{"subject": "...", "subject_type": "...", "relation": "...", "object": "...", "object_type": "..."}}]}}

Passage ({heading}):
{text}"""


class Extractor(Protocol):
    def extract(self, chunk: Chunk) -> list[ExtractedTriplet]: ...


def filter_strict(items: list[dict[str, Any]], chunk_id: str) -> list[ExtractedTriplet]:
    """Keep only schema types and relations; normalise casing; drop self-loops and empties."""
    kept: list[ExtractedTriplet] = []
    for item in items:
        subject = str(item.get("subject", "")).strip().strip("`")
        obj = str(item.get("object", "")).strip().strip("`")
        s_type = TYPE_BY_UPPER.get(str(item.get("subject_type", "")).strip().upper())
        o_type = TYPE_BY_UPPER.get(str(item.get("object_type", "")).strip().upper())
        relation = str(item.get("relation", "")).strip().upper().replace(" ", "_")
        if not subject or not obj or s_type is None or o_type is None:
            continue
        if relation not in LLM_RELATIONS or subject.lower() == obj.lower():
            continue
        if len(subject) > 80 or len(obj) > 80:
            continue
        kept.append(ExtractedTriplet(subject, s_type, relation, obj, o_type, chunk_id))
    return kept


class JsonExtractor:
    """Our own schema-guided prompt (fallback backend)."""

    def __init__(self, llm: LLMClient, max_triplets: int = 10, tag: str = "json") -> None:
        self.llm = llm
        self.max_triplets = max_triplets
        self.tag = tag

    def extract(self, chunk: Chunk) -> list[ExtractedTriplet]:
        prompt = EXTRACT_PROMPT.format(
            max_triplets=self.max_triplets, heading=" > ".join(chunk.heading_path), text=chunk.text
        )
        resp = self.llm.complete([{"role": "user", "content": prompt}], stage="extraction", tag=self.tag)
        try:
            data = extract_json(resp.text)
        except ValueError:
            return []
        items = data.get("triplets", [])
        if not isinstance(items, list):
            return []
        return filter_strict([i for i in items if isinstance(i, dict)], chunk.chunk_id)


def _cached_llama_llm(client: LLMClient, tag: str) -> Any:
    """A LlamaIndex CustomLLM that routes every completion through our cached client."""
    from llama_index.core.bridge.pydantic import PrivateAttr
    from llama_index.core.llms import CompletionResponse, CompletionResponseGen, CustomLLM, LLMMetadata
    from llama_index.core.llms.callbacks import llm_completion_callback

    class CachedLlamaLLM(CustomLLM):
        model_name: str = "tsunagu-cached"
        _client: Any = PrivateAttr()
        _tag: str = PrivateAttr()

        def __init__(self, inner: LLMClient, inner_tag: str) -> None:
            super().__init__()
            self._client = inner
            self._tag = inner_tag

        @property
        def metadata(self) -> LLMMetadata:
            return LLMMetadata(
                context_window=32000,
                num_output=2048,
                model_name=self.model_name,
                is_chat_model=False,
                is_function_calling_model=False,
            )

        @llm_completion_callback()
        def complete(self, prompt: str, formatted: bool = False, **kwargs: Any) -> CompletionResponse:
            resp = self._client.complete([{"role": "user", "content": prompt}], stage="extraction", tag=self._tag)
            return CompletionResponse(text=resp.text)

        @llm_completion_callback()
        def stream_complete(self, prompt: str, formatted: bool = False, **kwargs: Any) -> CompletionResponseGen:
            response = self.complete(prompt, formatted=formatted, **kwargs)

            def gen() -> CompletionResponseGen:
                yield response

            return gen()

    return CachedLlamaLLM(client, tag)


class LlamaIndexExtractor:
    """LlamaIndex `SchemaLLMPathExtractor`, strict, uppercase schema (see Task 16 note)."""

    def __init__(self, llm: LLMClient, max_triplets: int = 10, tag: str = "llamaindex") -> None:
        from llama_index.core.indices.property_graph import SchemaLLMPathExtractor

        upper_types = tuple(t.upper() for t in ENTITY_TYPES)
        self.extractor = SchemaLLMPathExtractor(
            llm=_cached_llama_llm(llm, tag),
            possible_entities=Literal[upper_types],  # type: ignore[valid-type,arg-type]
            possible_relations=Literal[LLM_RELATIONS],  # type: ignore[valid-type,arg-type]
            kg_validation_schema={t: list(LLM_RELATIONS) for t in upper_types},  # type: ignore[misc]
            strict=True,
            max_triplets_per_chunk=max_triplets,
            num_workers=1,
        )

    def extract(self, chunk: Chunk) -> list[ExtractedTriplet]:
        from llama_index.core.graph_stores.types import KG_NODES_KEY, KG_RELATIONS_KEY
        from llama_index.core.schema import TextNode

        [node] = self.extractor([TextNode(text=chunk.text, id_=chunk.chunk_id)])
        entities = {n.id: n for n in node.metadata.get(KG_NODES_KEY, [])}
        items: list[dict[str, Any]] = []
        for rel in node.metadata.get(KG_RELATIONS_KEY, []):
            subj, obj = entities.get(rel.source_id), entities.get(rel.target_id)
            if subj is None or obj is None:
                continue
            items.append(
                {"subject": subj.name, "subject_type": subj.label, "relation": rel.label, "object": obj.name, "object_type": obj.label}
            )
        return filter_strict(items, chunk.chunk_id)


def build_extractor(backend: str, llm: LLMClient, max_triplets: int, tag: str | None = None) -> Extractor:
    if backend == "llamaindex":
        return LlamaIndexExtractor(llm, max_triplets, tag or "llamaindex")
    if backend == "json":
        return JsonExtractor(llm, max_triplets, tag or "json")
    raise ValueError(f"unknown extraction backend {backend}")

class TableEmbedder:
    """Fixed vectors for known names; orthogonal one-hot vectors otherwise."""

    def __init__(self, table: dict[str, list[float]]) -> None:
        self.table = table
        self.next = 0

    def embed_documents(self, texts: list[str]) -> np.ndarray:
        rows = []
        for t in texts:
            if t in self.table:
                v = np.array(self.table[t] + [0.0] * 8, dtype=np.float32)
            else:
                v = np.zeros(10, dtype=np.float32)
                v[2 + self.next % 8] = 1.0
                self.next += 1
            rows.append(v / np.linalg.norm(v))
        return np.stack(rows)

    def embed_query(self, text: str) -> np.ndarray:
        return self.embed_documents([text])[0]

"""Entity resolution: normalise, alias map, symbol links, embedding merge, review sample."""

from __future__ import annotations

import csv
import random
import re
from collections import defaultdict
from dataclasses import asdict, dataclass, fields
from pathlib import Path

import yaml

from tsunagu.embeddings import Embedder
from tsunagu.graph import ENTITY_TYPES, Graph, Node

CODE_TYPES = ("Class", "Function", "Parameter")


@dataclass
class Merge:
    reason: str  # normalize | alias | symbol_link | embedding
    type: str
    a: str
    b: str
    score: float


def singularize(word: str) -> str:
    if len(word) <= 3 or word.endswith(("ss", "us", "is")):
        return word
    if word.endswith("ies"):
        return word[:-3] + "y"
    if word.endswith(("ches", "shes", "xes")):
        return word[:-2]
    if word.endswith("s"):
        return word[:-1]
    return word


def normalize_name(name: str) -> str:
    """Lowercase, strip backticks and `()`, collapse separators, singularise the last word."""
    s = name.strip().strip("`").strip().lower().removesuffix("()")
    s = re.sub(r"[\s_\-]+", " ", s).strip()
    if not s:
        return s
    words = s.split(" ")
    words[-1] = singularize(words[-1])
    return " ".join(words)


def load_aliases(path: Path) -> dict[str, str]:
    """`canonical: [aliases]` YAML to {normalised alias: normalised canonical}."""
    data = yaml.safe_load(path.read_text(encoding="utf-8")) or {}
    mapping: dict[str, str] = {}
    for canonical, aliases in data.items():
        for alias in aliases or []:
            mapping[normalize_name(str(alias))] = normalize_name(str(canonical))
    return mapping


class UnionFind:
    def __init__(self) -> None:
        self.parent: dict[str, str] = {}

    def find(self, x: str) -> str:
        self.parent.setdefault(x, x)
        while self.parent[x] != x:
            self.parent[x] = self.parent[self.parent[x]]
            x = self.parent[x]
        return x

    def union(self, a: str, b: str) -> bool:
        ra, rb = self.find(a), self.find(b)
        if ra == rb:
            return False
        self.parent[rb] = ra
        return True


def _apply(graph: Graph, uf: UnionFind) -> Graph:
    groups: dict[str, list[Node]] = defaultdict(list)
    for nid, node in graph.nodes.items():
        groups[uf.find(nid)].append(node)
    out = Graph()
    mapping: dict[str, str] = {}
    for members in groups.values():
        rep = max(members, key=lambda n: (n.type != "Symbol", len(n.chunk_ids), -len(n.name)))
        merged = Node(
            id=rep.id,
            name=rep.name,
            type=rep.type,
            title=rep.title,
            chunk_ids=set().union(*(m.chunk_ids for m in members)),
            aliases={m.name for m in members if m.name != rep.name}.union(*(m.aliases for m in members)),
        )
        out.nodes[merged.id] = merged
        for m in members:
            mapping[m.id] = merged.id
    for e in graph.edges:
        out.add_edge(mapping[e.src], e.rel, mapping[e.dst], e.source, e.chunk_id)
    return out


def resolve_graph(
    graph: Graph, aliases: dict[str, str], embedder: Embedder, threshold: float
) -> tuple[Graph, list[Merge]]:
    """Merge duplicate entities; `Page` nodes are never touched.

    Returns:
        The resolved graph and the log of every union performed.
    """
    uf = UnionFind()
    merges: list[Merge] = []
    nodes = [n for n in graph.nodes.values() if n.type != "Page"]

    def canon(n: Node) -> tuple[str, bool]:
        norm = normalize_name(n.name)
        return aliases.get(norm, norm), norm in aliases

    groups: dict[tuple[str, str], list[tuple[Node, bool]]] = defaultdict(list)
    for n in nodes:
        name, via_alias = canon(n)
        groups[(n.type, name)].append((n, via_alias))
    for (type_, _), members in groups.items():
        first, first_alias = members[0]
        for other, other_alias in members[1:]:
            if uf.union(first.id, other.id):
                reason = "alias" if first_alias or other_alias else "normalize"
                merges.append(Merge(reason, type_, first.name, other.name, 1.0))

    code_index: dict[str, str] = {}
    for type_ in CODE_TYPES:
        for n in nodes:
            if n.type == type_:
                code_index.setdefault(canon(n)[0], n.id)
    for n in nodes:
        if n.type == "Symbol":
            target = code_index.get(canon(n)[0])
            if target and uf.union(target, n.id):
                merges.append(Merge("symbol_link", "Symbol", graph.nodes[target].name, n.name, 1.0))

    for type_ in ENTITY_TYPES:
        reps = sorted({uf.find(n.id) for n in nodes if n.type == type_})
        if len(reps) < 2:
            continue
        vectors = embedder.embed_documents([graph.nodes[r].name for r in reps])
        sims = vectors @ vectors.T
        pairs = [
            (float(sims[i, j]), i, j)
            for i in range(len(reps))
            for j in range(i + 1, len(reps))
            if sims[i, j] > threshold
        ]
        for score, i, j in sorted(pairs, reverse=True):
            if uf.union(reps[i], reps[j]):
                merges.append(Merge("embedding", type_, graph.nodes[reps[i]].name, graph.nodes[reps[j]].name, score))
    return _apply(graph, uf), merges


def write_merges_csv(merges: list[Merge], path: Path) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("w", encoding="utf-8", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=[fl.name for fl in fields(Merge)])
        writer.writeheader()
        for m in sorted(merges, key=lambda m: (m.reason, -m.score, m.type, m.a)):
            writer.writerow(asdict(m))


def sample_for_review(merges: list[Merge], n: int = 30, seed: int = 42) -> list[Merge]:
    """Random sample of embedding merges (topped up with other non-trivial merges if too few)."""
    rng = random.Random(seed)
    embedding = [m for m in merges if m.reason == "embedding"]
    others = [m for m in merges if m.reason in ("alias", "symbol_link")]
    if len(embedding) >= n:
        return rng.sample(embedding, n)
    return embedding + rng.sample(others, min(n - len(embedding), len(others)))


def write_review_csv(sample: list[Merge], path: Path) -> None:
    """Review sheet: the owner writes y (correct merge) or n (false merge) in `correct`."""
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("w", encoding="utf-8", newline="") as f:
        writer = csv.writer(f, lineterminator="\n")
        writer.writerow(["reason", "type", "a", "b", "score", "correct"])
        for m in sample:
            writer.writerow([m.reason, m.type, m.a, m.b, f"{m.score:.3f}", ""])


def read_review(path: Path) -> tuple[int, int]:
    """Return (reviewed rows, false merges) from the review sheet."""
    with path.open(encoding="utf-8") as f:
        rows = list(csv.DictReader(f))
    marks = [r["correct"].strip().lower() for r in rows if r.get("correct", "").strip()]
    return len(marks), sum(1 for m in marks if m.startswith("n"))

```python
"""Retriever C: seed entities, expand along the graph, rank reached chunks by dense similarity."""

from __future__ import annotations

import re
import time
from collections import defaultdict
from typing import Any

import numpy as np

from tsunagu.embeddings import Embedder
from tsunagu.graph import ALL_RELATIONS, Graph
from tsunagu.resolve import normalize_name
from tsunagu.retrievers import RetrievalResult, to_retrieved
from tsunagu.store import ChunkStore

CODE_SPAN_RE = re.compile(r"`([^`]+)`")
WORD_RE = re.compile(r"[A-Za-z0-9_\-]+")
STOPWORDS = {
    "how", "what", "which", "when", "where", "why", "who", "does", "do", "the", "and", "for",
    "with", "can", "use", "using", "are", "is", "you", "your", "that", "this", "from", "into",
    "should", "must", "need", "there", "their", "they", "about", "between", "after", "before",
}


def query_terms(query: str) -> set[str]:
    """Normalised candidate entity names: code spans and word 1- to 3-grams."""
    terms = {normalize_name(m.group(1)) for m in CODE_SPAN_RE.finditer(query)}
    words = WORD_RE.findall(query)
    for n in (1, 2, 3):
        for i in range(len(words) - n + 1):
            gram = " ".join(words[i : i + n])
            if n == 1 and (gram.lower() in STOPWORDS or len(gram) < 3):
                continue
            terms.add(normalize_name(gram))
    terms.discard("")
    return terms


class GraphRetriever:
    def __init__(
        self,
        graph: Graph,
        store: ChunkStore,
        embedder: Embedder,
        *,
        chunk_matrix: np.ndarray,
        node_ids: list[str],
        node_matrix: np.ndarray,
        hops: int = 2,
        relations: set[str] | frozenset[str] | None = None,
        seed_vector_k: int = 5,
        seed_min_sim: float = 0.6,
        max_expand_degree: int = 40,
        name: str = "graph",
    ) -> None:
        self.graph = graph
        self.store = store
        self.embedder = embedder
        self.chunk_matrix = chunk_matrix
        self.node_ids = node_ids
        self.node_pos = {nid: i for i, nid in enumerate(node_ids)}
        self.node_matrix = node_matrix
        self.hops = hops
        self.relations = frozenset(relations) if relations is not None else ALL_RELATIONS
        self.seed_vector_k = seed_vector_k
        self.seed_min_sim = seed_min_sim
        self.max_expand_degree = max_expand_degree
        self.name = name
        self.name_index: dict[str, list[str]] = defaultdict(list)
        for node in graph.nodes.values():
            if node.type == "Page":
                continue
            for label in {node.name, *node.aliases}:
                self.name_index[normalize_name(label)].append(node.id)

    def find_seeds(self, query: str, q_vec: np.ndarray) -> list[tuple[str, str, float]]:
        """Seeds as (node_id, how, score): exact name matches first, then vector matches."""
        seeds: dict[str, tuple[str, float]] = {}
        for term in sorted(query_terms(query)):
            for nid in self.name_index.get(term, []):
                seeds.setdefault(nid, ("name", 1.0))
        sims = self.node_matrix @ q_vec
        for i in np.argsort(-sims)[: self.seed_vector_k]:
            if sims[i] >= self.seed_min_sim:
                seeds.setdefault(self.node_ids[i], ("vector", float(sims[i])))
        return [(nid, how, score) for nid, (how, score) in seeds.items()]

    def expand(self, seeds: list[str], q_vec: np.ndarray) -> tuple[dict[str, int], list[tuple[str, str, str, int]]]:
        """Breadth-first expansion, at most `max_expand_degree` neighbours per node."""
        hop_of = {s: 0 for s in seeds}
        frontier = list(seeds)
        edges: list[tuple[str, str, str, int]] = []
        for hop in range(1, self.hops + 1):
            next_frontier: list[str] = []
            for nid in frontier:
                neighbours = [(e, o) for e, o in self.graph.neighbors(nid, self.relations) if o not in hop_of]
                neighbours.sort(key=lambda eo: -self._node_sim(eo[1], q_vec))
                for edge, other in neighbours[: self.max_expand_degree]:
                    if other in hop_of:
                        continue
                    hop_of[other] = hop
                    next_frontier.append(other)
                    edges.append((edge.src, edge.rel, edge.dst, hop))
            frontier = next_frontier
        return hop_of, edges

    def _node_sim(self, nid: str, q_vec: np.ndarray) -> float:
        pos = self.node_pos.get(nid)
        return float(self.node_matrix[pos] @ q_vec) if pos is not None else 0.0

    def retrieve(self, query: str, k: int) -> RetrievalResult:
        started = time.perf_counter()
        q_vec = self.embedder.embed_query(query)
        seeds = self.find_seeds(query, q_vec)
        hop_of, edges = self.expand([s for s, _, _ in seeds], q_vec)
        pool: dict[str, int] = {}
        for nid, hop in hop_of.items():
            for cid in self.graph.nodes[nid].chunk_ids:
                if cid in self.store.position:
                    pool[cid] = min(pool.get(cid, hop), hop)
        scores = self.chunk_matrix @ q_vec
        ranked = sorted(pool, key=lambda cid: -scores[self.store.position[cid]])[:k]
        fallback = 0
        if len(ranked) < k:
            for i in np.argsort(-scores):
                cid = self.store.ids[i]
                if cid not in ranked:
                    ranked.append(cid)
                    fallback += 1
                if len(ranked) == k:
                    break
        chunks = [to_retrieved(self.store, cid, float(scores[self.store.position[cid]])) for cid in ranked]
        trace: dict[str, Any] = {
            "seeds": seeds,
            "edges": edges[:60],
            "pool_size": len(pool),
            "fallback": fallback,
            "hop_of_result": {cid: pool.get(cid) for cid in ranked},
        }
        return RetrievalResult(self.name, chunks, time.perf_counter() - started, trace)
```

from __future__ import annotations

import json
from functools import cached_property
from typing import Any

import numpy as np

from tsunagu.config import Settings
from tsunagu.embeddings import SentenceTransformerEmbedder
from tsunagu.graph import Graph

class Resources:
    def __init__(self, settings: Settings) -> None:
        self.s = settings

    @cached_property
    def store(self) -> ChunkStore:
        return ChunkStore.load(self.s.paths.processed_dir / "chunks.jsonl")

    @cached_property
    def embedder(self) -> SentenceTransformerEmbedder:
        return SentenceTransformerEmbedder(self.s.retrieval.embed_model)

    @cached_property
    def collection(self) -> Any:
        return get_collection(persistent_client(self.s.paths.index_dir))

    @cached_property
    def chunk_matrix(self) -> np.ndarray:
        ids = json.loads((self.s.paths.index_dir / "chunk_ids.json").read_text(encoding="utf-8"))
        if ids != self.store.ids:
            raise RuntimeError("index/chunk_ids.json does not match chunks.jsonl: run `tsunagu index`")
        return np.load(self.s.paths.index_dir / "chunk_embeddings.npy")

    @cached_property
    def graph(self) -> Graph:
        return Graph.load(self.s.paths.graph_dir / "graph_resolved.json")

    @cached_property
    def node_index(self) -> tuple[list[str], np.ndarray]:
        ids_path = self.s.paths.index_dir / "node_ids.json"
        mat_path = self.s.paths.index_dir / "node_embeddings.npy"
        node_ids = list(self.graph.nodes)
        if ids_path.exists() and mat_path.exists() and json.loads(ids_path.read_text(encoding="utf-8")) == node_ids:
            return node_ids, np.load(mat_path)
        matrix = self.embedder.embed_documents([self.graph.nodes[n].embed_text for n in node_ids])
        np.save(mat_path, matrix)
        ids_path.write_text(json.dumps(node_ids), encoding="utf-8")
        return node_ids, matrix


def build_retriever(name: str, settings: Settings, res: Resources) -> Retriever:
    """Build retriever `name` ("dense", "hybrid" or "graph")."""
    if name == "dense":
        return DenseRetriever(res.collection, res.store, res.embedder)
    if name == "hybrid":
        dense = DenseRetriever(res.collection, res.store, res.embedder)
        return HybridRetriever(dense, BM25Retriever(res.store), settings.retrieval.candidate_k, settings.retrieval.rrf_k)
    if name == "graph":
        node_ids, node_matrix = res.node_index
        g = settings.graph
        return GraphRetriever(
            res.graph,
            res.store,
            res.embedder,
            chunk_matrix=res.chunk_matrix,
            node_ids=node_ids,
            node_matrix=node_matrix,
            hops=g.hops,
            seed_vector_k=g.seed_vector_k,
            seed_min_sim=g.seed_min_sim,
            max_expand_degree=g.max_expand_degree,
        )
    raise ValueError(f"unknown system {name}; known: {', '.join(SYSTEMS)}")

```python
"""LLM judge: correctness (0/1/2), claim-level faithfulness, citation precision."""

from __future__ import annotations

from dataclasses import dataclass
from typing import Any

from tsunagu.answer import Answer, cited_pages, format_context
from tsunagu.llm import LLMClient, extract_json
from tsunagu.questions import Question
from tsunagu.retrievers import RetrievedChunk

CORRECTNESS_PROMPT = """You grade an answer to a question about the LangGraph documentation.

Question: {question}
Reference answer: {gold}
Candidate answer: {answer}

Rubric:
2 = correct and complete: states every key point of the reference answer and contradicts none.
1 = partially correct: at least one key point is right, but a key point is missing or slightly wrong.
0 = wrong, contradicts the reference, irrelevant, or says it cannot answer.
Extra correct detail that is not in the reference does not lower the score.

Reply with one JSON object only: {{"score": 0, "reason": "<one sentence>"}} where score is 0, 1 or 2."""

FAITHFULNESS_PROMPT = """Split the answer into atomic factual claims. For each claim, decide whether the context passages support it.

Context passages (each starts with its label in square brackets):
{context}

Answer:
{answer}

Reply with one JSON object only:
{{"claims": [{{"claim": "...", "supported": true, "support_labels": ["<label>"]}}]}}"""


@dataclass
class Judgement:
    correctness: int | None
    correctness_reason: str
    claims_total: int = 0
    claims_supported: int = 0
    cited_pages: int = 0
    precise_citations: int = 0


def judge_correctness(llm: LLMClient, question: Question, answer: str) -> tuple[int | None, str]:
    prompt = CORRECTNESS_PROMPT.format(question=question.question, gold=question.gold_answer, answer=answer)
    resp = llm.complete([{"role": "user", "content": prompt}], stage="judge", tag="correctness")
    try:
        data = extract_json(resp.text)
        score = int(data["score"])
    except (ValueError, KeyError, TypeError):
        return None, f"unparseable judge reply: {resp.text[:200]}"
    if score not in (0, 1, 2):
        return None, f"score out of range: {score}"
    return score, str(data.get("reason", ""))


def judge_faithfulness(llm: LLMClient, chunks: list[RetrievedChunk], answer: str) -> list[dict[str, Any]]:
    prompt = FAITHFULNESS_PROMPT.format(context=format_context(chunks), answer=answer)
    resp = llm.complete([{"role": "user", "content": prompt}], stage="judge", tag="faithfulness")
    try:
        claims = extract_json(resp.text).get("claims", [])
    except ValueError:
        return []
    return [c for c in claims if isinstance(c, dict)] if isinstance(claims, list) else []


def citation_precision(citations: list[str], gold_pages: list[str], claims: list[dict[str, Any]]) -> tuple[int, int]:
    """Return (cited pages, cited pages that are gold or support a supported claim)."""
    pages = cited_pages(citations)
    supporting = {
        str(label).strip("[] ").split("#", 1)[0]
        for claim in claims
        if claim.get("supported")
        for label in claim.get("support_labels", []) or []
    }
    precise = sum(1 for p in pages if p in gold_pages or p in supporting)
    return len(pages), precise


def judge_answer(llm: LLMClient, question: Question, chunks: list[RetrievedChunk], ans: Answer) -> Judgement:
    """Score one answer. Abstention cases are decided without calling the judge."""
    if question.type == "unanswerable":
        if ans.abstained:
            return Judgement(2, "correct abstention")
        return Judgement(0, "answered an unanswerable question")
    if ans.abstained:
        return Judgement(0, "false abstention")
    score, reason = judge_correctness(llm, question, ans.answer)
    claims = judge_faithfulness(llm, chunks, ans.answer)
    supported = sum(1 for c in claims if c.get("supported") is True)
    n_cited, n_precise = citation_precision(ans.citations, question.gold_pages, claims)
    return Judgement(score, reason, len(claims), supported, n_cited, n_precise)
```

import csv

from typing import Any

CSV_COLUMNS = [
    "qid", "type", "system", "repeat", "hit", "recall", "mrr", "correctness", "abstained",
    "parse_ok", "claims_total", "claims_supported", "cited_pages", "precise_citations",

]
EMPTY_ANSWER: dict[str, Any] = {
    "answer": None, "citations": [], "abstained": None, "parse_ok": None, "answer_latency_s": None,
    "answer_prompt_tokens": None, "answer_completion_tokens": None, "correctness": None,
    "judge_reason": "", "claims_total": None, "claims_supported": None, "cited_pages": None,
    "precise_citations": None,
}


def run_eval(
    questions: list[Question],
    retrievers: dict[str, Retriever],
    llm: LLMClient | None,
    *,
    repeats: int,
    k: int,
    retrieval_only: bool = False,
    log: Callable[[str], None] = lambda message: None,
) -> list[dict[str, Any]]:
    """Evaluate every (question, system); answer and judge `repeats` times unless retrieval_only."""
    rows: list[dict[str, Any]] = []
    for qi, q in enumerate(questions, start=1):
        for name, retriever in retrievers.items():
            result = retriever.retrieve(q.question, k)
            pages = page_ranking([c.page_path for c in result.chunks])
            answerable = q.type != "unanswerable"
            base: dict[str, Any] = {
                "qid": q.id,
                "type": q.type,
                "system": name,
                "question": q.question,
                "gold_answer": q.gold_answer,
                "gold_pages": q.gold_pages,
                "retrieved_pages": pages,
                "retrieved_chunks": [c.chunk_id for c in result.chunks],
                "hit": hit_at_k(pages, q.gold_pages) if answerable else None,
                "recall": recall_at_k(pages, q.gold_pages) if answerable else None,
                "mrr": mrr(pages, q.gold_pages) if answerable else None,
                "retrieval_latency_s": result.latency_s,
                "graph_pool_size": result.trace.get("pool_size"),
                "graph_fallback": result.trace.get("fallback"),
            }
            if retrieval_only or llm is None:
                rows.append({**base, "repeat": 0, **EMPTY_ANSWER})
                continue
            for r in range(repeats):
                ans = answer_question(llm, q.question, result.chunks, tag=f"{name}:{q.id}:r{r}")
                j = judge_answer(llm, q, result.chunks, ans)
                rows.append(
                    {
                        **base,
                        "repeat": r,
                        "answer": ans.answer,
                        "citations": ans.citations,
                        "abstained": ans.abstained,
                        "parse_ok": ans.parse_ok,
                        "answer_latency_s": ans.latency_s,
                        "answer_prompt_tokens": ans.prompt_tokens,
                        "answer_completion_tokens": ans.completion_tokens,
                        "correctness": j.correctness,
                        "judge_reason": j.correctness_reason,
                        "claims_total": j.claims_total,
                        "claims_supported": j.claims_supported,
                        "cited_pages": j.cited_pages,
                        "precise_citations": j.precise_citations,
                    }
                )
        log(f"{qi}/{len(questions)} questions done")
    return rows


def merge_rows(old: list[dict[str, Any]], new: list[dict[str, Any]]) -> list[dict[str, Any]]:
    """Replace rows of the systems present in `new`, keep the others."""
    rerun = {r["system"] for r in new}
    return [r for r in old if r["system"] not in rerun] + new


def write_rows(rows: list[dict[str, Any]], results_dir: Path, stem: str = "eval_rows") -> None:
    """Write `<stem>.jsonl` (full) and `per_question.csv` (flat, lists joined with ';')."""
    write_jsonl(results_dir / f"{stem}.jsonl", rows)
    if stem != "eval_rows":
        return
    with (results_dir / "per_question.csv").open("w", encoding="utf-8", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=CSV_COLUMNS, extrasaction="ignore")
        writer.writeheader()
        for row in rows:
            flat = {k: (";".join(v) if isinstance(v, list) else v) for k, v in row.items()}
            writer.writerow(flat)

"""Blind manual labels vs the LLM judge: agreement and Cohen's kappa."""

from __future__ import annotations

import csv
import random
from pathlib import Path
from typing import Any

from tsunagu.metrics import cohen_kappa


def make_blind_sample(
    rows: list[dict[str, Any]], n: int = 20, seed: int = 7
) -> tuple[list[dict[str, Any]], list[dict[str, Any]]]:
    """Pick n judged answers balanced across systems; return (label sheet, hidden key)."""
    rng = random.Random(seed)
    pool = [
        r for r in rows
        if r["repeat"] == 0 and r["type"] != "unanswerable" and not r["abstained"] and r["correctness"] is not None
    ]
    by_system: dict[str, list[dict[str, Any]]] = {}
    for r in pool:
        by_system.setdefault(r["system"], []).append(r)
    for items in by_system.values():
        rng.shuffle(items)
    picked: list[dict[str, Any]] = []
    order = sorted(by_system)
    while len(picked) < n and any(by_system[s] for s in order):
        for s in order:
            if by_system[s] and len(picked) < n:
                picked.append(by_system[s].pop())
    rng.shuffle(picked)
    labels, key = [], []
    for i, r in enumerate(picked, start=1):
        item = f"L{i:02d}"
        labels.append({"item": item, "question": r["question"], "gold_answer": r["gold_answer"], "answer": r["answer"], "your_score": ""})
        key.append({"item": item, "qid": r["qid"], "system": r["system"], "repeat": r["repeat"]})
    return labels, key


def write_csv(rows: list[dict[str, Any]], path: Path) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("w", encoding="utf-8", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=list(rows[0]))
        writer.writeheader()
        writer.writerows(rows)


def compute_calibration(labels_path: Path, key_path: Path, rows: list[dict[str, Any]]) -> dict[str, Any]:
    """Compare the owner's scores with the judge's on the labelled items."""
    with labels_path.open(encoding="utf-8") as f:
        labels = {r["item"]: r["your_score"].strip() for r in csv.DictReader(f)}
    with key_path.open(encoding="utf-8") as f:
        key = {r["item"]: r for r in csv.DictReader(f)}
    judge = {(r["qid"], r["system"], int(r["repeat"])): r["correctness"] for r in rows}
    human: list[int] = []
    model: list[int] = []
    disagreements: list[dict[str, Any]] = []
    for item, score in sorted(labels.items()):
        if score == "":
            continue
        k = key[item]
        j = judge[(k["qid"], k["system"], int(k["repeat"]))]
        human.append(int(score))
        model.append(int(j))
        if int(score) != int(j):
            disagreements.append({"item": item, "qid": k["qid"], "human": int(score), "judge": int(j)})
    if not human:
        return {"n": 0}
    pass_h = [int(h == 2) for h in human]
    pass_m = [int(m == 2) for m in model]
    return {
        "n": len(human),
        "agreement": sum(h == m for h, m in zip(human, model, strict=True)) / len(human),
        "kappa": cohen_kappa(human, model),
        "agreement_pass": sum(h == m for h, m in zip(pass_h, pass_m, strict=True)) / len(human),
        "kappa_pass": cohen_kappa(pass_h, pass_m),
        "disagreements": disagreements,
    }
```

"""Generate the numbers in results/results.md (AUTO block only) and factual observations."""

from __future__ import annotations

import math
from pathlib import Path
from typing import Any

from tsunagu.metrics import percentile, wilson_interval

TYPES = ("single_fact", "multi_hop", "relational", "exact_reference", "unanswerable")
AUTO_START = "<!-- AUTO:START (generated by `tsunagu report`; do not edit by hand) -->"
AUTO_END = "<!-- AUTO:END -->"
MANUAL_TEMPLATE = """

## Findings

_To be written by the owner from the tables above: 3 to 5 honest bullets, including where the graph lost._

## Limits

_To be written by the owner: corpus size, 30 questions, one LLM for extraction, answers and judge, judge bias._

## What I would do at Safran

_To be written by the owner: 5 lines._
"""


def _mean(values: list[float]) -> float:
    return sum(values) / len(values) if values else float("nan")


def _f(x: float | None, digits: int = 2) -> str:
    return "n/a" if x is None or (isinstance(x, float) and math.isnan(x)) else f"{x:.{digits}f}"


def _pct(x: float) -> str:
    return "n/a" if math.isnan(x) else f"{100 * x:.0f}%"


def _systems(rows: list[dict[str, Any]]) -> list[str]:
    return list(dict.fromkeys(r["system"] for r in rows))


def question_outcomes(rows: list[dict[str, Any]], system: str) -> dict[str, dict[str, Any]]:
    """Per question: majority pass (score 2), mean score, stability across repeats."""
    by_q: dict[str, dict[str, Any]] = {}
    for r in rows:
        if r["system"] != system or r["correctness"] is None:
            continue
        entry = by_q.setdefault(r["qid"], {"type": r["type"], "scores": []})
        entry["scores"].append(r["correctness"])
    for entry in by_q.values():
        scores = entry["scores"]
        entry["passed"] = sum(1 for s in scores if s == 2) > len(scores) / 2
        entry["mean_score"] = _mean(scores)
        entry["stable"] = len(set(scores)) == 1
    return by_q


def _pass_cell(outcomes: dict[str, dict[str, Any]], types: tuple[str, ...]) -> tuple[str, int, int]:
    selected = [o for o in outcomes.values() if o["type"] in types]
    k, n = sum(1 for o in selected if o["passed"]), len(selected)
    if n == 0:
        return "n/a", 0, 0
    lo, hi = wilson_interval(k, n)
    return f"{k}/{n} = {_pct(k / n)} [{_pct(lo)}, {_pct(hi)}]", k, n


def retrieval_section(rows: list[dict[str, Any]], systems: list[str]) -> str:
    base = [r for r in rows if r["repeat"] == 0 and r["type"] != "unanswerable"]
    lines = [
        "### Retrieval (hit@8 / recall@8 / MRR, page level)",
        "",
        "| Type | " + " | ".join(systems) + " |",
        "|---|" + "---|" * len(systems),
    ]
    for qtype in (*TYPES[:-1], "all"):
        sel = [r for r in base if qtype == "all" or r["type"] == qtype]
        n = len({r["qid"] for r in sel})
        cells = []
        for s in systems:
            mine = [r for r in sel if r["system"] == s]
            cells.append(
                f"{_f(_mean([r['hit'] for r in mine]))} / {_f(_mean([r['recall'] for r in mine]))} / {_f(_mean([r['mrr'] for r in mine]))}"
            )
        lines.append(f"| {qtype} (n={n}) | " + " | ".join(cells) + " |")
    return "\n".join(lines)


def correctness_section(rows: list[dict[str, Any]], systems: list[str]) -> str:
    outcomes = {s: question_outcomes(rows, s) for s in systems}
    lines = [
        "### Correctness (judge score 2 in a majority of repeats), 95% Wilson CI, n = questions",
        "",
        "| Type | " + " | ".join(systems) + " |",
        "|---|" + "---|" * len(systems),
    ]
    groups: list[tuple[str, tuple[str, ...]]] = [(t, (t,)) for t in TYPES]
    groups += [("answerable", TYPES[:-1]), ("all", TYPES)]
    for label, types in groups:
        lines.append(f"| {label} | " + " | ".join(_pass_cell(outcomes[s], types)[0] for s in systems) + " |")
    mean_cells = [_f(_mean([o["mean_score"] for o in outcomes[s].values()])) for s in systems]
    stable_cells = [_pct(_mean([float(o["stable"]) for o in outcomes[s].values()])) for s in systems]
    lines.append("| mean score (0 to 2) | " + " | ".join(mean_cells) + " |")
    lines.append("| repeat stability | " + " | ".join(stable_cells) + " |")
    return "\n".join(lines)


def generation_section(rows: list[dict[str, Any]], systems: list[str]) -> str:
    lines = [
        "### Faithfulness, citation precision, abstention (pooled over repeats)",
        "",
        "| Metric | " + " | ".join(systems) + " |",
        "|---|" + "---|" * len(systems),
    ]
    metrics: dict[str, list[str]] = {
        "faithfulness (supported claims)": [],
        "citation precision": [],
        "correct abstention (unanswerable)": [],
        "false abstention (answerable)": [],
        "JSON parse failures": [],
        "judge unparseable": [],
    }
    for s in systems:
        mine = [r for r in rows if r["system"] == s and r.get("parse_ok") is not None]
        judged = [r for r in mine if r["type"] != "unanswerable" and not r["abstained"]]
        claims = sum(r["claims_total"] or 0 for r in judged)
        supported = sum(r["claims_supported"] or 0 for r in judged)
        cited = sum(r["cited_pages"] or 0 for r in judged)
        precise = sum(r["precise_citations"] or 0 for r in judged)
        unans = [r for r in mine if r["type"] == "unanswerable"]
        answ = [r for r in mine if r["type"] != "unanswerable"]
        metrics["faithfulness (supported claims)"].append(f"{supported}/{claims} = {_pct(supported / claims if claims else float('nan'))}")
        metrics["citation precision"].append(f"{precise}/{cited} = {_pct(precise / cited if cited else float('nan'))}")
        metrics["correct abstention (unanswerable)"].append(_pct(_mean([float(bool(r["abstained"])) for r in unans])))
        metrics["false abstention (answerable)"].append(_pct(_mean([float(bool(r["abstained"])) for r in answ])))
        metrics["JSON parse failures"].append(str(sum(1 for r in mine if r["parse_ok"] is False)))
        metrics["judge unparseable"].append(str(sum(1 for r in mine if str(r.get("judge_reason", "")).startswith("unparseable"))))
    for name, cells in metrics.items():
        lines.append(f"| {name} | " + " | ".join(cells) + " |")
    return "\n".join(lines)


def ops_section(rows: list[dict[str, Any]], systems: list[str]) -> str:
    lines = [
        "### Latency and tokens per query",
        "",
        "| Metric | " + " | ".join(systems) + " |",
        "|---|" + "---|" * len(systems),
    ]
    cells: dict[str, list[str]] = {k: [] for k in ("retrieval p50 / p95 (ms)", "end-to-end p50 / p95 (s)", "answer tokens per query (in + out)", "graph candidate pool (mean chunks)", "graph dense fallback (chunks, total)")}
    for s in systems:
        mine = [r for r in rows if r["system"] == s and r["answer_latency_s"] is not None]
        retr = [r["retrieval_latency_s"] * 1000 for r in mine]
        e2e = [r["retrieval_latency_s"] + r["answer_latency_s"] for r in mine]
        tokens = [r["answer_prompt_tokens"] + r["answer_completion_tokens"] for r in mine]
        pools = [r["graph_pool_size"] for r in mine if r["graph_pool_size"] is not None]
        cells["retrieval p50 / p95 (ms)"].append(f"{_f(percentile(retr, 50), 0)} / {_f(percentile(retr, 95), 0)}")
        cells["end-to-end p50 / p95 (s)"].append(f"{_f(percentile(e2e, 50), 1)} / {_f(percentile(e2e, 95), 1)}")
        cells["answer tokens per query (in + out)"].append(_f(_mean([float(t) for t in tokens]), 0))
        cells["graph candidate pool (mean chunks)"].append(_f(_mean([float(p) for p in pools]), 0) if pools else "n/a")
        fallback = [r["graph_fallback"] for r in mine if r["graph_fallback"] is not None and r["repeat"] == 0]
        cells["graph dense fallback (chunks, total)"].append(str(sum(fallback)) if fallback else "n/a")
    for name, row in cells.items():
        lines.append(f"| {name} | " + " | ".join(row) + " |")
    return "\n".join(lines)

    return (

def render_auto(rows: list[dict[str, Any]], header: str, calib: dict[str, Any] | None, repeats: int) -> str:
    systems = _systems(rows)
    n_q = len({r["qid"] for r in rows})
    note = (
        f"Statistics: {n_q} questions, {repeats} repeat(s) per system. With about 30 questions, differences under "
        "about 15 points are probably noise; per-type n is 3 to 8, so per-type rates are indicative only."
    )
    parts = [header, retrieval_section(rows, systems), correctness_section(rows, systems)]
    parts += [generation_section(rows, systems), ops_section(rows, systems), calibration_section(calib), note]
    return "\n\n".join(parts)


def observations(rows: list[dict[str, Any]], systems: list[str]) -> str:
    """Facts only: best system per type and whether the Wilson intervals overlap."""
    outcomes = {s: question_outcomes(rows, s) for s in systems}
    lines = ["# Observations (generated facts, not conclusions)", ""]
    for qtype in (*TYPES, "answerable"):
        types = TYPES[:-1] if qtype == "answerable" else (qtype,)
        stats = []
        for s in systems:
            _, k, n = _pass_cell(outcomes[s], types)
            if n:
                stats.append((k / n, s, wilson_interval(k, n)))
        if len(stats) < 2:
            continue
        stats.sort(reverse=True)
        (best_rate, best, best_ci), (second_rate, second, second_ci) = stats[0], stats[1]
        overlap = best_ci[0] <= second_ci[1]
        verdict = "intervals overlap: not distinguishable at this n" if overlap else "intervals do not overlap"
        lines.append(
            f"- {qtype}: highest pass rate {best} {_pct(best_rate)}, next {second} {_pct(second_rate)}; {verdict}."
        )
    return "\n".join(lines) + "\n"


def write_results(path: Path, auto_md: str) -> None:
    """Replace the AUTO block in results.md, creating the file with owner sections if missing."""
    block = f"{AUTO_START}\n\n{auto_md}\n\n{AUTO_END}"
    if path.exists():
        text = path.read_text(encoding="utf-8")
        if AUTO_START in text and AUTO_END in text:
            head, rest = text.split(AUTO_START, 1)
            _, tail = rest.split(AUTO_END, 1)
            path.write_text(head + block + tail, encoding="utf-8")
            return
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text("# Tsunagu results\n\n" + block + MANUAL_TEMPLATE, encoding="utf-8")
```

```python
class GraphHybridRetriever:
    """Union of hybrid and graph results, reranked with RRF, top k."""

    def __init__(self, hybrid: HybridRetriever, graph: GraphRetriever, rrf_k: int = 60) -> None:
        self.hybrid = hybrid
        self.graph = graph
        self.rrf_k = rrf_k
        self.name = "graph_hybrid"

    def retrieve(self, query: str, k: int) -> RetrievalResult:
        started = time.perf_counter()
        h = self.hybrid.retrieve(query, k)
        g = self.graph.retrieve(query, k)
        fused = rrf_fuse([[c.chunk_id for c in h.chunks], [c.chunk_id for c in g.chunks]], k=self.rrf_k)[:k]
        chunks = [to_retrieved(self.hybrid.dense.store, cid, score) for cid, score in fused]
        return RetrievalResult(self.name, chunks, time.perf_counter() - started, g.trace)
